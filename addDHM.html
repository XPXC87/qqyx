禮包碼保存系統

※ 資料已與雲端資料庫同步

const CLOUDFLARE_API = 'https://xpxc.qzz.io/api'; // 網頁開啟時自動同步雲端資料 window.onload = async function() { const textarea = document.getElementById('codes'); try { const response = await fetch(CLOUDFLARE_API); if (response.ok) { const dbCodes = await response.json(); if (dbCodes && dbCodes.length > 0) { textarea.value = dbCodes.join('\n'); } } } catch (error) { console.log("雲端無紀錄或同步失敗"); } }; async function submit() { const textarea = document.getElementById('codes'); const content = textarea.value.trim(); if (!content) return alert("請輸入禮包碼"); // 提示正在處理 const btn = document.querySelector('button'); const originalText = btn.innerText; btn.innerText = "同步並提交中..."; btn.disabled = true; try { // 只發送一次請求給 Worker，由 Worker 處理後續所有動作 const response = await fetch(CLOUDFLARE_API, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ code: content }) }); if (response.ok) { alert("提交成功！資料已同步至雲端並發送至後端。"); } else { alert("提交失敗，請檢查 Worker 狀態。"); } } catch (error) { alert("提交錯誤：" + error.message); } finally { btn.innerText = originalText; btn.disabled = false; } }
